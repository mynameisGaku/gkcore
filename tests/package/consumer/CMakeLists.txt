cmake_minimum_required(VERSION 3.20)
project(gkcorePackageConsumer LANGUAGES CXX)
option(GKCORE_PACKAGE_GPU_SMOKE "Run an installed-SDK drawing smoke test on the host GPU" OFF)
find_package(gkcore CONFIG REQUIRED)
add_executable(package_consumer main.cpp)
target_link_libraries(package_consumer PRIVATE gkcore::gkcore)
if(GKCORE_PACKAGE_GPU_SMOKE)
    target_compile_definitions(package_consumer PRIVATE GKCORE_PACKAGE_GPU_SMOKE=1)
endif()
